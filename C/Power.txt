#include <stdio.h>


int Os(int base, int power)
{
    if (power == 0)
        return 1;

    return base * Os(base, power - 1);
}

int main()
{
    int base, power;

    printf("Enter base: \n");
    fflush(stdout);
    scanf("%d", &base);

    printf("Enter power: \n");
    fflush(stdout);
    scanf("%d", &power);

    int result = Os(base, power);

    printf("Result = %d\n", result);

    return 0;
}
