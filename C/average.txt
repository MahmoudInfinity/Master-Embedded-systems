#include <stdio.h>

int main() {

	int n;
	double arr[100];
	double sum = 0.0;

	printf("Enter number of elements: \n");
	scanf("%d", &n);

	if (n <= 0|| n > 100) {
		printf("Invalid size.\n");
		return 1;
	}


	for (int i = 0; i < n; i++) {
		printf("Enter element %d: \n", i + 1);
		fflush(stdout);
		scanf("%lf", &arr[i]);
		sum += arr[i];
	}


	double average = sum / n;

	printf("Average = %.2lf\n", average);

	return 0;
}
