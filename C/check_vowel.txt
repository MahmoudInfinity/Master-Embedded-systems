#include <stdio.h>



int main() {

	unsigned char c ;

	printf("##########Console-output###\n");

	while(1)
	{
		printf("Enter the character you want  to check: \n");

		fflush(stdout);

		scanf(" %c", &c);

		if (c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u' ||
				c == 'A' || c == 'E' || c == 'I' || c == 'O' || c == 'U')
			printf("%c is a vowel.\n" , c);
		else
			printf("%c is not a vowel.\n" , c);



	}
	return 0;
}
