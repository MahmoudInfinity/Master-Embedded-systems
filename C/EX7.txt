#include <stdio.h>



int main() {

	float x , y  ;

	printf("##########Console-output###\n");

	printf("Enter the value of a: \n");

	fflush(stdout);

	scanf("%f",&x);

	printf("Enter the value of b: \n");

	fflush(stdout);

	scanf("%f",&y);

     // x = 1 , y =2

	x = x + y;

	// x= 3

	y = x - y;

	// y = (1 + 2) - 2 = 1


	x = x - y;

	// x = (1 + 2) - 1 = 2




	printf("After swapping value of a  =   %f\n", x);
	printf("After swapping value of b =   %f\n", y);

	printf("###########################\n"
			"########################################################################");

	return 0;
}
