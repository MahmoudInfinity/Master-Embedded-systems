#include <stdio.h>
#include <string.h>

int main() {

    char str[100];
    int i = 0, count = 0;

    printf("Enter a string to get its length: \n");
    fflush(stdout);
    fgets(str, sizeof(str), stdin);
    str[strcspn(str, "\n")] = '\0';


    while(str[i] != '\0')
    {

            count++;
            i++;
    }

    printf("length  of the string is: %d\n",  count);

    return 0;
}
