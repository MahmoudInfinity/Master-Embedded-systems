#include <stdio.h>



int main() {

	unsigned char x   ;

	printf("##########Console-output###\n");
	printf("This program checks if the character is alphabet \n");

	while(1)
	{


		printf("Enter the the character you want to check : \n");

		fflush(stdout);

		scanf(" %c", &x);


		if ((x >= 65 && x<= 90)|| (x >= 97 && x <= 122))

			printf("The Character is alphabet \n");
		else

			printf("The Character is NOT alphabet \n");

	}
	return 0;

}
