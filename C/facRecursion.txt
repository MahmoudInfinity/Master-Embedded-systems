#include <stdio.h>

int  factorial(int n )
{

	if(n == 1)
		return n;


	return n * factorial(n - 1);



 }




int main()

{
	int n , fac;

	printf("Please enter a number to calculate its factorial: \n");

	fflush(stdout);

	scanf("%d", &n);

	printf(" \n");

	fac = factorial(n);

	printf("factorial of %d is %d \n", n , fac);



}
